# ─────────────────────────────────────────────
# Helpers
# ─────────────────────────────────────────────

def load_model(variable: str):
    path = os.path.join('saved_models', f'sarima_{variable.lower()}.pkl')

    if not os.path.exists(path):
        raise FileNotFoundError(
            f"Model not found: {path}. Run training first."
        )

    return joblib.load(path)


def compute_ces(rainfall, temp_mean, humidity, baseline_stats: dict):

    z_rain = (
        (rainfall - baseline_stats['rain_mean']) /
        (baseline_stats['rain_std'] + 1e-8)
    )

    z_temp = (
        (temp_mean - baseline_stats['temp_mean']) /
        (baseline_stats['temp_std'] + 1e-8)
    )

    z_hum = (
        (humidity - baseline_stats['hum_mean']) /
        (baseline_stats['hum_std'] + 1e-8)
    )

    ces = z_rain + z_temp + z_hum

    def label_z(z):
        az = abs(z)

        if az >= 2:
            return 'Strong'
        elif az >= 1.5:
            return 'Moderate'
        else:
            return 'Normal'

    def label_ces(c):
        ac = abs(c)

        if ac >= 3:
            return 'Extreme'
        elif ac >= 1:
            return 'Moderate'
        else:
            return 'Normal'

    return {
        'Z_Rainfall': round(z_rain, 3),
        'Rain_Label': label_z(z_rain),

        'Z_Temp': round(z_temp, 3),
        'Temp_Label': label_z(z_temp),

        'Z_Humidity': round(z_hum, 3),
        'Hum_Label': label_z(z_hum),

        'CES': round(ces, 3),
        'Climate_Condition': label_ces(ces)
    }


def forecast_from_model(model, steps: int = 12):
    fc = model.get_forecast(steps=steps)
    return fc.predicted_mean


# ─────────────────────────────────────────────
# Main
# ─────────────────────────────────────────────

def run_test(csv_path: str = None, steps: int = 12):

    print("\n" + "=" * 60)
    print("  Bangalore Climate Analytics — Model Testing")
    print("=" * 60)

    # -----------------------------------------
    # Load dataset
    # -----------------------------------------

    data_path = csv_path if csv_path else 'data.csv'

    print(f"\nLoading data from: {data_path}")

    df = pd.read_csv(data_path)

    # Rename columns
    df.columns = ['Date', 'Rainfall', 'Temp_Max', 'Temp_Min', 'Humidity']

    # Convert date
    df['Date'] = pd.to_datetime(
        df['Date'],
        errors='coerce',
        dayfirst=True
    )

    # -----------------------------------------
    # CLEAN DATA
    # -----------------------------------------

    # Replace special rainfall values
    df['Rainfall'] = df['Rainfall'].replace('Tr', 0)

    # Convert numeric columns
    numeric_cols = [
        'Rainfall',
        'Temp_Max',
        'Temp_Min',
        'Humidity'
    ]

    for col in numeric_cols:
        df[col] = pd.to_numeric(df[col], errors='coerce')

    # Remove invalid rows
    df = df.dropna(subset=numeric_cols)

    # Create mean temperature
    df['Temp_mean'] = (
        df['Temp_Max'] + df['Temp_Min']
    ) / 2

    # Extract year and month
    df['Year'] = df['Date'].dt.year
    df['Month'] = df['Date'].dt.month

    # -----------------------------------------
    # Monthly Aggregation
    # -----------------------------------------

    monthly = df.groupby(['Year', 'Month']).agg(
        Rainfall=('Rainfall', 'sum'),
        Temp_mean=('Temp_mean', 'mean'),
        Humidity=('Humidity', 'mean')
    ).reset_index()

    # -----------------------------------------
    # Baseline statistics
    # -----------------------------------------

    baseline_stats = {
        'rain_mean': monthly['Rainfall'].mean(),
        'rain_std': monthly['Rainfall'].std(),

        'temp_mean': monthly['Temp_mean'].mean(),
        'temp_std': monthly['Temp_mean'].std(),

        'hum_mean': monthly['Humidity'].mean(),
        'hum_std': monthly['Humidity'].std(),
    }

    print("\nBaseline Statistics:")
    for k, v in baseline_stats.items():
        print(f"{k}: {round(v, 2)}")

    # -----------------------------------------
    # Load models
    # -----------------------------------------

    print("\nLoading models...")

    models = {}

    for var in ['Rainfall', 'Temp_mean', 'Humidity']:

        try:
            models[var] = load_model(var)
            print(f"Loaded model: {var}")

        except Exception as e:
            print(f"WARNING: {e}")

    if not models:
        print("\nNo models found.")
        return

    # -----------------------------------------
    # Forecast
    # -----------------------------------------

    print(f"\nGenerating {steps}-month forecast...")

    forecasts = {}

    for var, model in models.items():
        forecasts[var] = forecast_from_model(
            model,
            steps=steps
        )

    forecast_df = pd.DataFrame(forecasts)

    print("\nForecasted Values:")
    print(forecast_df.round(2).to_string())

    # -----------------------------------------
    # CES Analysis
    # -----------------------------------------

    print("\nClimate Condition Analysis:")

    ces_results = []
    for idx, (i, row) in enumerate(forecast_df.iterrows(), start=1):

        rain = row.get(
            'Rainfall',
            baseline_stats['rain_mean']
        )

        temp = row.get(
            'Temp_mean',
            baseline_stats['temp_mean']
        )

        hum = row.get(
            'Humidity',
            baseline_stats['hum_mean']
        )

        ces_info = compute_ces(
            rain,
            temp,
            hum,
            baseline_stats
        )

        # Use integer month counter
        ces_info['Month'] = idx

        ces_results.append(ces_info)
    # for i, row in forecast_df.iterrows():

    #     rain = row.get(
    #         'Rainfall',
    #         baseline_stats['rain_mean']
    #     )

    #     temp = row.get(
    #         'Temp_mean',
    #         baseline_stats['temp_mean']
    #     )

    #     hum = row.get(
    #         'Humidity',
    #         baseline_stats['hum_mean']
    #     )

    #     ces_info = compute_ces(
    #         rain,
    #         temp,
    #         hum,
    #         baseline_stats
    #     )

    #     ces_info['Month'] = i + 1

    #     ces_results.append(ces_info)

    ces_df = pd.DataFrame(ces_results)

    print(
        ces_df[
            [
                'Month',
                'Z_Rainfall',
                'Z_Temp',
                'Z_Humidity',
                'CES',
                'Climate_Condition'
            ]
        ].to_string(index=False)
    )

    # -----------------------------------------
    # Summary
    # -----------------------------------------

    print("\nForecast Summary:")
    print(
        ces_df['Climate_Condition']
        .value_counts()
        .to_string()
    )

    # -----------------------------------------
    # Plotting
    # -----------------------------------------

    os.makedirs('outputs', exist_ok=True)

    fig, axes = plt.subplots(4, 1, figsize=(12, 14))

    variables = ['Rainfall', 'Temp_mean', 'Humidity']
    colors = ['royalblue', 'tomato', 'seagreen']

    for ax, var, color in zip(
        axes[:3],
        variables,
        colors
    ):

        if var in forecast_df.columns:

            ax.plot(
                range(len(forecast_df)),
                forecast_df[var].values,
                color=color,
                marker='o'
            )

            ax.set_title(f'Forecast: {var}')
            ax.set_xlabel('Month Ahead')
            ax.set_ylabel(var)
            ax.grid(True)

    # CES plot
    axes[3].bar(
        range(len(ces_df)),
        ces_df['CES'].values,
        color='purple',
        alpha=0.7
    )

    axes[3].axhline(
        3,
        color='red',
        linestyle='--',
        label='Extreme'
    )

    axes[3].axhline(
        -3,
        color='red',
        linestyle='--'
    )

    axes[3].axhline(
        1,
        color='orange',
        linestyle=':',
        label='Moderate'
    )

    axes[3].axhline(
        -1,
        color='orange',
        linestyle=':'
    )

    axes[3].set_title(
        'Forecasted Climate Extreme Score (CES)'
    )

    axes[3].set_xlabel('Month Ahead')
    axes[3].set_ylabel('CES')

    axes[3].legend()
    axes[3].grid(True)

    plt.suptitle(
        f'Bangalore Climate Forecast — Next {steps} Months',
        fontsize=14
    )

    plt.tight_layout()

    out_path = 'outputs/test_forecast_output.png'

    plt.savefig(out_path, dpi=100)

    plt.close()

    print(f"\nForecast plot saved: {out_path}")

    # -----------------------------------------
    # Save results
    # -----------------------------------------

    csv_out = 'outputs/test_forecast_results.csv'

    ces_df.to_csv(csv_out, index=False)

    print(f"Results CSV saved: {csv_out}")

    print("\n" + "=" * 60)
    print("  Testing Complete!")
    print("=" * 60)

    parser = argparse.ArgumentParser(

        default=None,
        help='Path to CSV dataset'
    )

    parser.add_argument(
        '--steps',
        type=int,
        default=12,
        help='Forecast horizon'
    )

    args = parser.parse_args()

    run_test(
        csv_path=args.csv,
        steps=args.steps
